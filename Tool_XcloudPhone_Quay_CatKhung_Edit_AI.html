def split_into_subtitle_chunks(paragraph, max_chars=65):
    """Tách đoạn văn ~1000 ký tự thành từng câu phụ đề ngắn gọn để hiển thị trên video."""
    sentences = re.split(r'(?<=[.!?;,])\s+', paragraph.strip())
    chunks = []
    current = ""
    for sent in sentences:
        words = sent.split()
        for w in words:
            if len(current) + len(w) + 1 <= max_chars:
                current = f"{current} {w}".strip()
            else:
                if current:
                    chunks.append(current)
                current = w
        if current:
            chunks.append(current)
            current = ""
    return chunks if chunks else [paragraph.strip()[:max_chars]]

def get_vietnamese_font(font_size=28):
    """Tự động lấy font hỗ trợ tiếng Việt có sẵn trên Windows/Mac/Linux."""
    candidates = [
        "C:/Windows/Fonts/arialbd.ttf",
        "C:/Windows/Fonts/segoeuib.ttf",
        "C:/Windows/Fonts/tahomabd.ttf",
        "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf",
        "/System/Library/Fonts/Supplemental/Arial Bold.ttf",
    ]
    for path in candidates:
        if os.path.exists(path):
            return ImageFont.truetype(path, font_size)
    return ImageFont.load_default()

def crop_free_fire_region(frame_bgr):
    """Cắt lấy đúng vùng khung đang chơi Free Fire (có dư nhẹ viền) và đóng khung TikTok 9:16 nếu bật."""
    h, w, _ = frame_bgr.shape
    if ENABLE_CROP_FREE_FIRE:
        x1 = max(0, int(w * CROP_X_RATIO))
        y1 = max(0, int(h * CROP_Y_RATIO))
        cw = min(w - x1, int(w * CROP_W_RATIO))
        ch = min(h - y1, int(h * CROP_H_RATIO))
        cropped = frame_bgr[y1:y1 + ch, x1:x1 + cw]
    else:
        cropped = frame_bgr

    if TIKTOK_VERTICAL_9_16:
        out_w, out_h = 720, 1280
        bg = cv2.resize(cropped, (out_w, out_h))
        bg = cv2.GaussianBlur(bg, (35, 35), 0)
        bg = (bg.astype(np.float32) * 0.38).astype(np.uint8)
        ch_c, cw_c, _ = cropped.shape
        fg_w = out_w
        fg_h = int((ch_c / max(1, cw_c)) * fg_w)
        fg = cv2.resize(cropped, (fg_w, fg_h))
        y_off = int((out_h - fg_h) * 0.40)
        bg[y_off:y_off + fg_h, 0:fg_w] = fg
        return bg

    ch_f, cw_f, _ = cropped.shape
    return cropped[:ch_f - (ch_f % 2), :cw_f - (cw_f % 2)]

def draw_subtitle_on_frame(frame_bgr, subtitle_text, font):
    """Vẽ phụ đề tiếng Việt màu vàng viền đen có hộp mờ chuẩn CapCut/TikTok."""
    if not subtitle_text:
        return frame_bgr
    h, w, _ = frame_bgr.shape
    img_pil = Image.fromarray(cv2.cvtColor(frame_bgr, cv2.COLOR_BGR2RGB))
    draw = ImageDraw.Draw(img_pil, "RGBA")

    wrapped_lines = textwrap.wrap(subtitle_text, width=42)[:2]
    line_heights = []
    line_widths = []
    for line in wrapped_lines:
        bbox = draw.textbbox((0, 0), line, font=font)
        line_widths.append(bbox[2] - bbox[0])
        line_heights.append(bbox[3] - bbox[1] + 10)

    box_w = max(line_widths, default=200) + 44
    box_h = sum(line_heights) + 24
    box_x1 = max(14, (w - box_w) // 2)
    box_y1 = int(h * 0.82) - (box_h // 2)
    box_x2 = min(w - 14, box_x1 + box_w)
    box_y2 = min(h - 14, box_y1 + box_h)

    draw.rounded_rectangle(
        [box_x1, box_y1, box_x2, box_y2],
        radius=14,
        fill=(10, 15, 28, 215),
        outline=(250, 204, 21, 185),
        width=2,
    )

    curr_y = box_y1 + 12
    for idx, line in enumerate(wrapped_lines):
        lw = line_widths[idx]
        lx = (w - lw) // 2
        for dx, dy in [(-2, -2), (2, -2), (-2, 2), (2, 2), (0, 2)]:
            draw.text((lx + dx, curr_y + dy), line, font=font, fill=(0, 0, 0, 255))
        draw.text((lx, curr_y), line, font=font, fill=(250, 204, 21, 255))
        curr_y += line_heights[idx]

    return cv2.cvtColor(np.array(img_pil), cv2.COLOR_RGB2BGR)

async def generate_female_ai_voice(text, out_mp3_path, target_duration_sec=70.0):
    """Tạo giọng đọc AI Nữ Hoài My và tự canh tốc độ để khớp độ dài ~1000+ ký tự."""
    char_len = len(text.strip())
    estimated_sec = char_len / 14.5
    rate_str = "+0%"
    if estimated_sec > target_duration_sec * 0.95:
        speed_pct = min(45, max(5, int(((estimated_sec / (target_duration_sec * 0.92)) - 1.0) * 100)))
        rate_str = f"+{speed_pct}%"
    communicate = edge_tts.Communicate(text=text.strip(), voice=VOICE_NAME, rate=rate_str)
    await communicate.save(out_mp3_path)

Hỏi số lượng video muốn quay (VD: 5, 10, 20), tự động cắt đúng vùng khung Free Fire (hoặc full màn hình) và quay đúng 70 giây (1 phút 10 giây) mỗi video rồi ngắt: 

import os
import time
from datetime import datetime
import cv2
import numpy as np
import mss

DURATION_SECONDS = 70  # Đúng 1 phút 10 giây (70s)
FPS = 30
OUTPUT_DIR = "Video_XcloudPhone_FreeFire"

# Tọa độ cắt vừa khung đang chơi Free Fire (có dư nhẹ viền XcloudPhone):
CROP_ENABLED = True
CROP_X_RATIO = 0.085
CROP_Y_RATIO = 0.135
CROP_W_RATIO = 0.705
CROP_H_RATIO = 0.755

def main():
    print("=" * 64)
    print("  TOOL QUAY XCLOUDPHONE FREE FIRE - ĐÚNG 1 PHÚT 10 GIÂY + CẮT KHUNG")
    print("=" * 64)
    print("Lưu ý: Hãy mở sẵn màn hình XcloudPhone & điện thoại Free Fire!")
    print()

    while True:
        raw = input("👉 Mày muốn quay bao nhiêu video? (VD: 5, 10): ").strip()
        if raw.isdigit() and int(raw) > 0:
            total_videos = int(raw)
            break
        print("❌ Vui lòng nhập một số nguyên lớn hơn 0 (ví dụ: 5 hoặc 10)!")

    os.makedirs(OUTPUT_DIR, exist_ok=True)
    print(f"\n✅ Đã nhận lệnh: Quay {total_videos} video (Mỗi video đúng 1 phút 10 giây = 70s).")
    print("⏳ Đếm ngược 3 giây để mày chuyển sang màn hình XcloudPhone...")
    for c in range(3, 0, -1):
        print(f"   Bắt đầu sau {c} giây...")
        time.sleep(1)

    with mss.mss() as sct:
        mon = sct.monitors[1]
        if CROP_ENABLED:
            capture_region = {
                "left": mon["left"] + int(mon["width"] * CROP_X_RATIO),
                "top": mon["top"] + int(mon["height"] * CROP_Y_RATIO),
                "width": int(mon["width"] * CROP_W_RATIO),
                "height": int(mon["height"] * CROP_H_RATIO),
            }
        else:
            capture_region = mon

        width = capture_region["width"] - (capture_region["width"] % 2)
        height = capture_region["height"] - (capture_region["height"] % 2)

        for idx in range(1, total_videos + 1):
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = os.path.join(
                OUTPUT_DIR,
                f"XcloudPhone_FreeFire_Video_{idx:02d}_{timestamp}_1m10s.mp4"
            )
            fourcc = cv2.VideoWriter_fourcc(*"mp4v")
            out = cv2.VideoWriter(filename, fourcc, FPS, (width, height))

            print(f"\n🔴 [Video {idx}/{total_videos}] Đang quay khung Free Fire: {filename}")
            total_frames = DURATION_SECONDS * FPS
            start_time = time.perf_counter()

            for frame_idx in range(total_frames):
                target_time = start_time + (frame_idx / FPS)
                img = np.array(sct.grab(capture_region))
                frame = cv2.cvtColor(img, cv2.COLOR_BGRA2BGR)
                frame = frame[:height, :width]
                out.write(frame)

                elapsed = time.perf_counter() - start_time
                remaining = max(0.0, DURATION_SECONDS - elapsed)
                mins = int(remaining) // 60
                secs = int(remaining) % 60
                print(
                    f"\r   ⏱️ Thời gian còn lại: {mins:02d}:{secs:02d} "
                    f"(Đã quay {min(DURATION_SECONDS, int(elapsed))}s / {DURATION_SECONDS}s)",
                    end="",
                    flush=True,
                )

                sleep_time = target_time + (1.0 / FPS) - time.perf_counter()
                if sleep_time > 0:
                    time.sleep(sleep_time)

            out.release()
            print(f"\n✅ [Hoàn tất {idx}/{total_videos}] Đã lưu đúng 1 phút 10 giây: {filename}")

            if idx < total_videos:
                time.sleep(1)

    print("\n" + "=" * 64)
    print(f"🎉 XONG! Đã quay đủ {total_videos} video (1 phút 10 giây/video)!")
    print(f"📂 Thư mục chứa video: {os.path.abspath(OUTPUT_DIR)}")
    print("=" * 64)

if __name__ == "__main__":
    main()
